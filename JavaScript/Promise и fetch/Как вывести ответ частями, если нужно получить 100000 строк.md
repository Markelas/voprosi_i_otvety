**ReadableStream** — это интерфейс Web API для потокового чтения данных. Он позволяет получать данные частями по мере их поступления, вместо ожидания полной загрузки всего ответа.

## Основные концепции

- **Чанк (chunk)** — кусок данных (обычно Uint8Array или строка)
    
- **Reader** — объект для чтения чанков из потока
    
- **Controller** — управляет потоком (не виден напрямую)
    

## Чтение ответа от сервера частями

### Способ 1: Использование `response.body` (самый простой)
```js
async function readStreamResponse() {
    const response = await fetch('/api/large-data');
    
    // Получаем ReadableStream из тела ответа
    const reader = response.body.getReader();
    const decoder = new TextDecoder('utf-8');
    
    let result = '';
    let lineCount = 0;
    
    while (true) {
        // Читаем следующий чанк
        const { done, value } = await reader.read();
        
        // Если поток завершен - выходим
        if (done) break;
        
        // Декодируем бинарные данные в строку
        const chunk = decoder.decode(value, { stream: true });
        
        // Разбиваем на строки
        const lines = chunk.split('\n');
        for (const line of lines) {
            if (line) {
                lineCount++;
                // Обновляем UI для каждой строки
                updateUI(line, lineCount);
            }
        }
        
        result += chunk;
    }
    
    console.log(`Всего строк: ${lineCount}`);
}

function updateUI(line, count) {
    // Добавляем строку в DOM
    const div = document.createElement('div');
    div.textContent = `#${count}: ${line}`;
    document.getElementById('output').appendChild(div);
    
    // Автопрокрутка
    const container = document.getElementById('output');
    container.scrollTop = container.scrollHeight;
}
```
### Способ 2: Обработка потоковых JSON (NDJSON - Newline Delimited JSON)
```js
async function streamNDJSON() {
    const response = await fetch('/api/stream-json');
    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    
    let buffer = '';
    let lineNumber = 0;
    
    while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        
        // Декодируем и добавляем в буфер
        buffer += decoder.decode(value, { stream: true });
        
        // Обрабатываем строки
        const lines = buffer.split('\n');
        buffer = lines.pop() || ''; // Последняя часть может быть неполной
        
        for (const line of lines) {
            if (line.trim()) {
                try {
                    const data = JSON.parse(line);
                    lineNumber++;
                    renderData(data, lineNumber);
                } catch (e) {
                    console.warn('Ошибка парсинга строки:', line);
                }
            }
        }
    }
}
```

### Способ 3: С использованием `for await...of` (современный подход)

```js
async function streamWithForAwait() {
    const response = await fetch('/api/large-data');
    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    let lines = [];
    
    // Потоковая обработка через async iterator
    while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        
        const chunk = decoder.decode(value);
        const chunkLines = chunk.split('\n');
        
        // Обрабатываем полученные строки

            if (line) {
                lines.push(line);
                // Можно добавлять на страницу по мере поступления

            }
        }
    }
}
```

### Способ 4: Обработка с ограничением скорости (throttling)
```js
async function streamWithThrottle() {
    const response = await fetch('/api/large-data');
    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    
    const BATCH_SIZE = 100; // Показываем по 100 строк
    let batch = [];
    let totalLines = 0;
    
    while (true) {
        const { done, value } = await reader.read();
        if (done) {
            // Выводим остаток
            if (batch.length) renderBatch(batch);
            break;
        }
        
        const chunk = decoder.decode(value);
        const lines = chunk.split('\n');
        
        for (const line of lines) {
            if (line) {
                batch.push(line);
                totalLines++;
                
                // Если набрали батч - выводим
                if (batch.length >= BATCH_SIZE) {
                    renderBatch(batch);
                    batch = [];
                    
                    // Даем браузеру отдохнуть и обновиться
                    await new Promise(resolve => setTimeout(resolve, 0));
                }
            }
        }
    }
    
    console.log(`Обработано строк: ${totalLines}`);
}

function renderBatch(lines) {
    const fragment = document.createDocumentFragment();
    lines.forEach(line => {
        const div = document.createElement('div');
        div.textContent = line;
        fragment.appendChild(div);
    });
    document.getElementById('output').appendChild(fragment);
}
```
